let libraryName = "City Library";

let totalCheckouts = 0;               
const libraryConfig = {                
  location: "Main Street",
};
 
const membershipRules = {              
  finePerDay: 0.5,
  maxRenewals: 2,
};
 
class Media {
  #isCheckedOut = false;
  #id;
 
  constructor(title, genre) {
    if (new.target === Media) {
      throw new Error("Media is abstract and cannot be instantiated directly.");
    }
    this.title = title;
    this.genre = genre;
    this.#id = Math.floor(Math.random() * 100000);
  }
 
  getInfo() {
    throw new Error("getInfo() must be implemented by subclass");
  }
 
  checkOut() {
    if (!this.#isCheckedOut) {
      this.#isCheckedOut = true;
      totalCheckouts++;
      return true;
    } else {
      return false;
    }
  }
 
  returnItem() {
    this.#isCheckedOut = false;
  }
 
  isAvailable() {
    return !this.#isCheckedOut;
  }
 
  getId() {
    return this.#id;
  }
}
 
class Book extends Media {
  constructor(title, genre, author, pages) {
    super(title, genre); 
    this.author = author;
    this.pages = pages;
  }
 
  getInfo() {
    return `📖 "${this.title}" by ${this.author} (${this.pages} pages) [${this.genre}]`;
  }
}
 
class Movie extends Media {
  constructor(title, genre, director, duration) {
    super(title, genre);
    this.director = director;
    this.duration = duration;
  }
 
  getInfo() {
    return `🎬 "${this.title}" directed by ${this.director} (${this.duration} min) [${this.genre}]`;
  }
}
 
class Library {
  #catalog = [];
 
  constructor(name) {
    this.name = name;
    this.loans = []; 
  }
 

  addItem(item) {
    this.#catalog.push(item);
  }
 
  listAvailableItems() {
    const available = []; 
 
    for (let i = 0; i < this.#catalog.length; i++) {
      const item = this.#catalog[i];
    
      if (item.isAvailable()) {
        available.push(item.getInfo());
      }
    }
    return available;
  }
 
  checkOutItem(title) {
    let found = null;
 
    for (const item of this.#catalog) {
      if (item.title === title) {
        found = item;
        break;
      }
    }
 
    if (found && found.checkOut()) {
      this.loans.push(found.getId());
      return `✅ You checked out: ${found.getInfo()}`;
    } else if (found) {
      return `❌ "${title}" is already checked out.`;
    } else {
      return `⚠️ "${title}" not found in catalog.`;
    }
  }
 
  printCatalogSummary() {
    console.log(`--- ${this.name} Catalog Summary ---`);
    console.log(`Hours: ${libraryConfig.openHours}, Location: ${libraryConfig.location}`);
 
    let index = 0;
    while (index < this.#catalog.length) {
      const item = this.#catalog[index];
  
      const status = item.isAvailable() ? "Available" : "Checked Out";
      console.log(`${index + 1}. ${item.getInfo()} - ${status}`);
      index++;
    }
  }
}
